Non-DL baselines on the medoid_64 splits

Date: 11/09/2026

GOHREP

Goal:

Put a floor under the deep models, using the same hand-crafted features, the same model families and the same code as Chapter 2’s 10b_score_prediction.qmd, but evaluated on the train/val/test split every model in this repository is evaluated on.

Rationale:

This is a port, not a reimplementation. Sections 1-6 reproduce 10b_score_prediction.qmd line for line – the same twelve features, the same balanced undersample, the same source-grouped five-fold cross validation, the same maxnodes sweep, the same proportional-odds ordinal regression, the same baselines and the same leakage check – so the numbers here can be checked against the ones that analysis published. Section 7 is the only new part: the identical models refit on the medoid_64 train split, tuned on val and scored once on the 308 test CDAs, which is what makes them comparable with models/03 and models/04.

Written in R for the same reason the original is: MASS::polr fits the proportional-odds model, and there is no equivalent in this project’s Python environment.

The features: eleven segmentation features plus entropy_raw. frac_g_gt_r is deliberately NOT a model feature – its Otsu-artefact behaviour is discussed in 10_segmentation_and_feature_extraction section 5.2 – and is kept only for the single-feature “Threshold: CDA area” baseline.

# 2.3  Leakage sanity check: shuffled labels MUST collapse to chance.
rf_reg_cv <- function(ntree, target = y) {
  pred <- numeric(length(target))
  for (f in 1:K) {
    tr <- fold != f
    te <- fold == f
    m <- randomForest::randomForest(x = X[tr, , drop = FALSE], y = target[tr], ntree = ntree)
    pred[te] <- predict(m, X[te, , drop = FALSE])
  }
  pmin(pmax(round(pred), 0), 6)
}
# Cached: five 200-tree forests on shuffled labels, and the answer cannot change unless
# the features or the folds do.
leak_cache <- file.path(analysis_dir, "model_cache", "leakage_check.rds")
if (file.exists(leak_cache)) {
  shuffle_acc <- readRDS(leak_cache)
} else {
  set.seed(7)
  y_shuffled <- sample(y)
  shuffle_acc <- mean(rf_reg_cv(200, y_shuffled) == y_shuffled)
  saveRDS(shuffle_acc, leak_cache)
}
cat(sprintf(
  "Leakage check - shuffled-label cross validation exact accuracy = %.3f (chance = %.3f)\n",
  shuffle_acc, 1 / 7
))

NTREE <- 200
rf_fit_cv <- function(maxnodes) {
  pred <- numeric(length(y))
  for (f in 1:K) {
    tr <- fold != f
    te <- fold == f
    m <- randomForest::randomForest(
      x = X[tr, , drop = FALSE], y = y[tr], ntree = NTREE, maxnodes = maxnodes
    )
    pred[te] <- predict(m, X[te, , drop = FALSE])
  }
  pmin(pmax(round(pred), 0), 6)
}
rf_fit_train <- function(maxnodes) {
  set.seed(1)
  m <- randomForest::randomForest(x = X, y = y, ntree = NTREE, maxnodes = maxnodes)
  pmin(pmax(round(predict(m, X)), 0), 6)
}
maxnodes_grid <- c(4, 8, 16, 32, 64, 128, 256)

4. Ordinal regression

Proportional-odds ordinal logistic regression: one linear predictor over the twelve features plus six ordered cut-points, so the model uses the ordering of the scale rather than treating the seven scores as unrelated labels.

Code 
# 4.1
ord_cv <- function() {
  pred <- integer(length(y))
  for (f in 1:K) {
    tr <- fold != f
    te <- fold == f
    mu <- colMeans(X[tr, ])
    sdv <- apply(X[tr, ], 2, sd)
    sdv[sdv == 0] <- 1
    dtr <- data.frame(scale(X[tr, , drop = FALSE], mu, sdv),
      y = factor(y[tr], levels = 0:6, ordered = TRUE)
    )
    dte <- data.frame(scale(X[te, , drop = FALSE], mu, sdv))
    fit <- suppressWarnings(MASS::polr(y ~ ., data = dtr, Hess = TRUE))
    pred[te] <- as.integer(as.character(predict(fit, dte, type = "class")))
  }
  pred
}
ord_train <- function() {
  mu <- colMeans(X)
  sdv <- apply(X, 2, sd)
  sdv[sdv == 0] <- 1
  d <- data.frame(scale(X, mu, sdv), y = factor(y, levels = 0:6, ordered = TRUE))
  fit <- suppressWarnings(MASS::polr(y ~ ., data = d, Hess = TRUE))
  as.integer(as.character(predict(fit, data.frame(scale(X, mu, sdv)), type = "class")))
}
ord_best <- ord_cv()
ord_tr <- ord_train()
cat(sprintf(
  "Ordinal regression: held-out exact = %.3f, near-miss = %.3f; training = %.3f, gap = %.3f\n",
  mean(ord_best == y), mean(abs(ord_best - y) <= 1),
  mean(ord_tr == y), mean(ord_tr == y) - mean(ord_best == y)
))

thresh_cv <- function(x) {
  pred <- integer(length(y))
  for (f in 1:K) {
    tr <- fold != f
    te <- fold == f
    med <- tapply(x[tr], y[tr], median)
    sc <- as.integer(names(med))
    o <- order(sc); med <- med[o]; sc <- sc[o]
    cuts <- cummax((head(med, -1) + tail(med, -1)) / 2)
    pred[te] <- sc[findInterval(x[te], cuts) + 1]
  }
  pred
}
thresh_train <- function(x) {
  med <- tapply(x, y, median)
  sc <- as.integer(names(med))
  o <- order(sc); med <- med[o]; sc <- sc[o]
  cuts <- cummax((head(med, -1) + tail(med, -1)) / 2)
  sc[findInterval(x, cuts) + 1]
}

zfit <- function(A) { mu <- colMeans(A); s <- apply(A, 2, sd); s[s == 0] <- 1; list(mu = mu, s = s) }
zap  <- function(A, p) scale(A, p$mu, p$s)

# Ordinal forest (Hornung 2020, J. Classification 37, 4-17, doi:10.1007/s00357-018-9302-x).
# A regression forest whose target is a set of SCORE VALUES optimised by out-of-bag
# performance, rather than the class values 0-6 assumed equally spaced and then rounded.
# The polr cut-points in section 9 are not equally spaced, so that assumption is measurably
# wrong on these crops, which is the reason for using it here.
of_fit <- function(Xa, ya, seed = 42) {
  set.seed(seed)
  ordinalForest::ordfor(depvar = "y",
                        data = data.frame(Xa, y = factor(ya, levels = 0:6, ordered = TRUE)),
                        perffunction = "equal")
}
of_predict <- function(fit, Xb) {
  as.integer(as.character(predict(fit, newdata = data.frame(Xb))$ypred))
}
ord_at <- function(Xa, ya, Xb) {
  p <- zfit(Xa)
  dtr <- data.frame(zap(Xa, p), y = factor(ya, levels = 0:6, ordered = TRUE))
  fit <- suppressWarnings(MASS::polr(y ~ ., data = dtr, Hess = TRUE))
  as.integer(as.character(predict(fit, data.frame(zap(Xb, p)), type = "class")))
}

# The same train+val fit, applied to the ENSEMBLE split as well. analyses/14 selects and
# tunes its combiners on that split and reports test once, so it needs both. The ensemble
# crops are held out of every model in this repository, so scoring them here is not
# leakage; they are simply a second unseen split.
# Class probabilities as well as the hard score, so analyses/14 can put this model into
# probability-averaging rules on the same footing as the softmax of a network.
ord_probs_at <- function(Xa, ya, Xb) {
  p <- zfit(Xa)
  dtr <- data.frame(zap(Xa, p), y = factor(ya, levels = 0:6, ordered = TRUE))
  fit <- suppressWarnings(MASS::polr(y ~ ., data = dtr, Hess = TRUE))
  pr <- predict(fit, data.frame(zap(Xb, p)), type = "probs")
  colnames(pr) <- paste0("p", 0:6)
  pr
}

8. Ordinal forest: regularisation sweep and importance

The same sweep the regression forest gets in section 3, on the ordinal forest’s capacity parameter. ordfor passes min.node.size through to ranger: a larger minimum node size means smaller trees, so it plays the role maxnodes plays there, in the opposite direction. Training accuracy is resubstitution; held-out is the medoid_64 validation split.

8b. Confusion matrix as a change from the human baseline

The model and human matrices look alike because both are band-diagonal. Differencing them makes the departure legible: each cell is the model’s row proportion minus the human row proportion, so red means the model does that more often than a rater does and blue less. Counts are not comparable directly (917 rater judgements against 308 model predictions), so both are row-normalised first.
